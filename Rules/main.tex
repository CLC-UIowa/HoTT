\subsection{Judgments}

\begin{definition}[Judgments.]
  Judgements in MLTT are of the form:
  \begin{enumerate}
  \item $\WfType \Gamma A$
  \item $\WfTerm \Gamma a A$
  \item $\EqType \Gamma A B$
  \item $\EqTerm \Gamma a b A$
  \end{enumerate}
\end{definition}
\begin{itemize}
\item $\mathcal{J}$ is the judgment thesis,
  ${\mathcal{J} \in \{a\co A, A ~ type, M \doteq N : \sigma, A \doteq B ~ type\}}$
\item $\Gamma \vdash \mathcal{J}$ is the judgment.
\end{itemize}

\begin{definition}[Context.]
  A \emph{context} is a finite list of variable declarations
  \[ x_1 : A_1, x_2 : A_2(x_1), x_3 : A_3(x_1, x_2), ..., x_n : A_n (x_1, ..., x_n) \]
  satisfying the condition that for each $1 \leq k \leq n$
  we can derive the judgment
  \[ x_1 : A_1, x_2 : A_2(x_1), x_3 : A_3(x_1, x_2), ... \vdash x_n : A_n (x_1, ..., x_n) \]
\end{definition}

\subsection{Type Families}

\begin{definition}[Family of Types.]
  $B(x)$ is a family of types over $A$ in the context $\Gamma$
  \[ \WfType {\Gamma, x \co A} {B(x)} \]
\end{definition}

\begin{definition}[Section.]
  The element of type $B(x)$ in the context $\Gamma, x : A$ is called a \emph{section} of the family $B$ over $A$ in the context $\Gamma$
  \[ \WfTerm {\Gamma, x \co A} {b(x)} {B(x)} \]
  Alternatively, $b(x)$ is an element of type $B(x)$ \emph{indexed} by
  $x\co A$ in the context $\Gamma$
\end{definition}

\subsection{The Empty Type}
\begin{definition}[Empty type ($\EmptyTy$)]
  Constructors (none)\\

  Induction Principle
  \[\texttt{ind}_\EmptyTy : \PiTy {x : \EmptyTy} P (x) \]
\end{definition}

Special case of induction principle
\[ \texttt{ex-falso} := \texttt{ind}_\EmptyTy : \EmptyTy \to A \]

\begin{definition}[Negation]
  \[ \neg A := A \to \EmptyTy \]

  $A$ is empty if it comes equipped with an element of type $\neg A$.

  \[ \texttt{is-empty}(A) := A \to \Empty \]
\end{definition}

\begin{remark}
  \begin{enumerate}
  \item Proof of negation: Assume $A$ to prove $\neg A$ and hence
    derive a contradiction
  \item Proof by contradiction: A proof technique which uses double negation elimination
  \item Double Negation Elimination: $\neg\neg P \then P$
  \end{enumerate}
\end{remark}

\begin{proposition}[Contrapositive]
  For any two types $P$ and $Q$, there is a function
\[ (P \to Q) \to (\neg Q \to \neg P)\]
\end{proposition}
\[ \Lam {f : P \to Q} {\Lam {\tilde q : Q \to \EmptyTy} {\Lam {p : P} {f (p)}}} : (P \to Q) \to  \neg Q \to \neg P\]

What is equality in a type theory from the perspective of types as
proof-relevant propositions?

If $p : x = y$ then $p$ is a \emph{witness} that the elements x and y

In a proof-relevant world, $p$ is not unique.

\subsection{The inductive definition of identity types}

\begin{definition}[Identity Type]
  An inductive family of types $a =_A x$ index by $x : A$ of which the
  constructor is \[ \Refl a : {a =_A a} \]
\end{definition}
The induction principle of identity types postulates that for any
family of types $P(x, p)$, indexed by $x : A$ and $p : a =_A x $ there
is a function:
\[
\texttt{ind-eq}_{a} : P \big( a , \Refl{a} \big) \to \PiTy {x : A} {\PiTy {p : a =_A x} {P(x, p)}}
\]

satisfying the equation

\[ \texttt{ind-eq}_a \big (u, a, \Refl{a} \big) \doteq u \]
where $ u : P \big(a, \Refl{a} \big)$

Terminology:
\begin{itemize}
\item An element of type $a =_A x$ is called an \emph{identification} of
  $a$ with $x$, or a \emph{path} from $a$ to $x$ The
\item The induction principle is called the \emph{identification
    elimination} or the \emph{path induction}.
\item Identity type on $A$ is is written $\IdTy{A}{}$
\end{itemize}
\begin{remark}
  Identity type is not just an inductive type, like
  the inductive types $\NatTy$, $\EmptyTy$ and $\UnitTy$; for example,
  but it is an inductive family of types.

  The constructor only provides an element $\Refl{a} : {a =_A a}$.

  The induction principle says that in order to prove something about
  all identifications of $a$ with some $x : A$, it suffices to prove
  it only for the base case $\Refl{a}$.
\end{remark}
