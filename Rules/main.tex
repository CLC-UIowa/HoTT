\section{Dependent Type Theory}

\subsection{Judgments}

\begin{definition}[Judgments.]
  Judgements in MLTT are of the form:
  \begin{enumerate}
  \item $\WfType \Gamma A$
  \item $\WfTerm \Gamma a A$
  \item $\EqType \Gamma A B$
  \item $\EqTerm \Gamma a b A$
  \end{enumerate}
\end{definition}
\begin{itemize}
\item $\mathcal{J}$ is the judgment thesis,
  ${\mathcal{J} \in \{a\co A, A ~ type, M \doteq N : \sigma, A \doteq B ~ type\}}$
\item $\Gamma \vdash \mathcal{J}$ is the judgment.
\end{itemize}

\begin{definition}[Context.]
  A \emph{context} is a finite list of variable declarations
  \[ x_1 : A_1, x_2 : A_2(x_1), x_3 : A_3(x_1, x_2), ..., x_n : A_n (x_1, ..., x_n) \]
  satisfying the condition that for each $1 \leq k \leq n$
  we can derive the judgment
  \[ x_1 : A_1, x_2 : A_2(x_1), x_3 : A_3(x_1, x_2), ... \vdash x_n : A_n (x_1, ..., x_n) \]
\end{definition}

\subsection{Type Families}

\begin{definition}[Family of Types.]
  $B(x)$ is a family of types over $A$ in the context $\Gamma$
  \[ \WfType {\Gamma, x \co A} {B(x)} \]
\end{definition}

\begin{definition}[Section.]
  The element of type $B(x)$ in the context $\Gamma, x : A$ is called a \emph{section} of the family $B$ over $A$ in the context $\Gamma$
  \[ \WfTerm {\Gamma, x \co A} {b(x)} {B(x)} \]
  Alternatively, $b(x)$ is an element of type $B(x)$ \emph{indexed} by
  $x\co A$ in the context $\Gamma$
\end{definition}

\item The formation of contexts, types and their elements~\cref{fig:formation-rules}.

\item Variable conversion
\item Substitution
\item Weakening
\item The generic element
\end{enumerate}

\subsection{Ordinary Function types}
\begin{remark}[Type Theory Extension]
  For a derivation $\trule{D}$
  \[ \irule[\trule{D}]
    {\mathcal{D}};
    {\WfTerm \Gamma a A}
  \]
  where $\mathcal{D}$ uses premises $\mathcal{H}_1$, $\mathcal{H}_2$,
  $\mathcal{H}_3$,  ... ,$\mathcal{H}_n$.
  the notation
  \[
    \irule[]
    {\irule[]{\mathcal{D}}; {\WfTerm \Gamma a A}};
    {\WfTerm \Gamma {c := a} A}
  \]
  is a short hand for extending the type theory with the constant $c$
  for which the following inference rules are valid:
  \begin{gather*}
    \ib{\irule[]{\mathcal{H}_1 \quad \mathcal{H}_2 ~\dots~ \mathcal{H}_n}; {\WfTerm \Gamma c A}}
    \rsp
    \ib{\irule[]{\mathcal{H}_1\quad \mathcal{H}_2~\dots~\mathcal{H}_n}; {\EqTerm \Gamma c a A}}
  \end{gather*}
\end{remark}

\begin{remark}
  A special case of dependent function types where the binder does not
  appear free in the term.
  \[ A \to B := \PiTy {x\co A} B \]
\end{remark}

\begin{gather*}
  \ib{\irule[\trule{$\to$}]
    {\WfType \Gamma A}{\WfType \Gamma B};
    {\WfType \Gamma {A \to B}}
  }
  \rsp
  \ib{\irule[\trule{\EqR\to}]
    {\WfType \Gamma A}{\WfType \Gamma B};
    {\EqType \Gamma {A \to B}{\PiTy {x\co A} B}}
  }
  \\
  \ib{\irule[\trule{$\lambda$}]
    {\WfType \Gamma B}{\WfTerm {\Gamma,x \co A } {b(x)} {B}};
    {\WfTerm \Gamma {\Lam x b(x)}{A \to B}}
  }
  \rsp
  \ib{\irule[\trule{ev}]
    {\WfTerm \Gamma f {A \to B}};
    {\WfTerm {\Gamma, x \co A} {f(x)} B}
  }
  \\
  \ib{\irule[\trule{$\beta\lambda$}]
    {\WfType \Gamma B}{\WfTerm {\Gamma,x \co A } {b(x)} {B}};
    {\EqTerm \Gamma {(\Lam y b(y)) x} {b (x)} {B}}
  }
  \rsp
  \ib{\irule[\trule{$\eta\lambda$}]
     {\WfTerm \Gamma f {A \to B}};
     {\EqTerm \Gamma {\Lam x {f(x)}} {f} {A \to B}}
  }
\end{gather*}

\begin{definition}[Identity Function]
  \[\texttt{id}_A : A \to A \]
  \[\texttt{id}_A := \Lam x x\]
\end{definition}

\begin{remark}
  Difference between $f(x,y)$, $f\Ap (x)\Ap (y)$ and $f\Ap (x\Ap (y))$

\begin{definition}[Composition function]
  \[\comp : (B \to C) \to (A \to B) \to (A \to C)\]
  \[\comp := \Lam g {\Lam f {\Lam x (g\Ap (f\Ap x))}}\]
\end{definition}
Notation  $g \circ f$ means $ \comp\Ap (g, f)$

\begin{lemma}[Composition of Functions Associative]
  The following judgment is derivable in the system
\[
  \irule[]
  {\WfTerm \Gamma f {A \to B}}
  {\WfTerm \Gamma g {B \to C}}
  {\WfTerm \Gamma h {C \to D}};
  {\EqTerm \Gamma {(h \circ g) \circ f} {h \circ (g \circ f)} {A \to D}}
\]
\end{lemma}

  \ib{\irule[\trule{$0_\NatTy$-ind}]
    {\WfType {\Gamma,n\co\NatTy} {P(n)}}
    {\WfTerm \Gamma {p_0} {P(0_\NatTy)}}
    {\WfTerm \Gamma {p_S} {\PiTy {n\co\NatTy} {P(n)} \to P(\Suc (n))}};
    {\EqTerm \Gamma {\Ind\NatTy{p_o, p_S, 0_\NatTy}} {p_0} {P(0_\NatTy)}}

  \\
  \ib{\irule[\trule{$\Suc$-ind}]
    {\WfType {\Gamma,n\co\NatTy} {P(n)}}
    {\WfTerm \Gamma {p_0} {P(0_\NatTy)}}
    {\WfTerm \Gamma {p_S} {\PiTy {n\co\NatTy} {P(n)} \to P(\Suc (n))}};
    {\EqTerm {\Gamma, n\co\NatTy} {\Ind\NatTy{p_o, p_S, \Suc(n)}} {p_S(n, \Ind\NatTy{p_o,p_S,n})} {P(\Suc(n))}}
  }
\end{gather*}

\begin{remark}[Uniqueness of Types]
Absence of \emph{uniqueness} of types for all elements in a type
theory amounts to rapidly descending into madness.
\end{remark}

\begin{remark}[Alternative Induction Principle]
  \[\irule[]
    {\WfType {\Gamma,n\co\NatTy} {P(n)}};
    {\WfTerm \Gamma {\ensuremath{\texttt{ind}_\NatTy}} {P(0_\NatTy) \to
        \Big( \big(\PiTy {n\co\NatTy} {P(n)} \to P(\Suc(n)) \big)
              \to \PiTy  {n\co\NatTy} {P(n)} \Big)}}
  \]
\end{remark}

\subsection{Addition of Natural Numbers}

\begin{definition}[$\AddNat(m,n)$]
  \[ \AddNat : \NatTy \to \NatTy \to \NatTy \]
  With the following specification:
  \[ \AddNat (m, 0_\NatTy) \doteq m \]
  \[ \AddNat (m, \Suc n) \doteq \Suc (\AddNat (m, n)) \]

  Notation $m + n$ means $\AddNat (m, n)$
\end{definition}
