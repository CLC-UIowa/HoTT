\begin{enumerate}[label=(\alph*)]
  \item Give a derivation for the following \textbf{element conversion rule}:
  \[
    \ib{\irule
      {\EqType{\Gamma} {A} {A'}}
      {\WfTerm \Gamma a A};
      {\WfTerm {\Gamma} {a} {A'}}}
  \]
  \item Give a derivation for the following \textbf{congruence rule} for element conversion:
  \[
    \ib{\irule
      {\EqType \Gamma {A} {A'}}
      {\EqTerm \Gamma {a} {b} {A}};
      {\EqTerm \Gamma {a} {b} {A'}}
    }
  \]
\end{enumerate}

\subsubsection*{Solution (a)}
%\newcommand\Subst[2]{[#2/#1]}
\begin{gather*}\hspace{-10em}
{\small
    \ib{\irule[\trule{S}]
      {\WfTerm \Gamma a A \qquad
        \irule[\trule{var-conv}]
        {\irule[\trule{sym}]{\EqType \Gamma {A} {A'}}; {\EqType \Gamma {A'} {A}}}
        {\irule[\trule{var}]{};{\Gamma, x : A' \vdash x : A'}};
        {\WfTerm {\Gamma , x : A} x {A'}}};
      {\WfTerm {\Gamma} {\Subst x a x} {\Subst x a A'}}}}
\end{gather*}

\subsubsection*{Solution (b)}

\begin{gather*}
    \ib{\irule[\trule{cong-subst}]
      {\EqTerm \Gamma a b A}
      {\irule[above]{\vdots};{\WfTerm {\Gamma , x : A} x A'}};
      {\EqTerm \Gamma {\Subst x a x} {\Subst x b x} {\Subst x a A'}}
    }
\end{gather*}

\section*{2.1}

The $\eta$-rule is often seen as a judgmental extensionality principle. Use the $\eta$-rule to show that if $f$ and $g$ take equal values, then they must be equal, i.e., give a derivation of the rule:

\begin{gather*}
\ib{\irule
    {\WfTerm \Gamma f {\PiTy {x\co A} {B(x)}}}
    {\WfTerm \Gamma g {\PiTy {x\co A} {B(x)}}}
    {\EqTerm {\Gamma, x \co A} {f(x)} {g(x)} {B(x)}};
    {\EqTerm \Gamma f g {\PiTy {x : A} {B(x)}}}}
\end{gather*}

\subsection*{Solution}
First let's derive a 3-ary transitivity rule. Call it trans$_{3}$:

\begin{gather*}
\small
\ib{\irule[\trule{trans}]
  {\irule[\trule{trans}]
    {\EqTerm \Gamma a b T} {\EqTerm \Gamma b c T};
    {\EqTerm \Gamma a c T}}
  {\EqTerm \Gamma c d T};
  {\EqTerm \Gamma a d T}}
\end{gather*}

Then invoke trans$_{3}$ on the follow three equalities:

\begin{gather*}
\footnotesize
\ib{\irule[\trule{trans$_{3}$}]
    {
      {\irule[\trule{\EqR\eta}]
        {\irule[\trule{sym}]
          {\WfTerm \Gamma f {\Pi_{x : A}B(x)}};
          {\EqTerm \Gamma {\lambda x. f(x)} f {\Pi_{x : A}B(x)}}};
        {\EqTerm \Gamma f {\lambda x. f(x)} {\Pi_{x : A}B(x)}}}
    \quad
      {\irule[\trule{\EqR{\lambda}}]
        {\EqTerm {\Gamma , x : A} {f(x)} {g(x)} {B(x)}};
        {\EqTerm \Gamma {\lambda x. f(x)}{\lambda x. g(x)} {\Pi_{x : A}B(x)}}}
    \quad
      {\irule[\trule{\EqR{\eta}}]
        {\WfTerm \Gamma g {\Pi_{x : A}B(x)}};
        {\EqTerm \Gamma {\lambda x. g(x)} {g} {\Pi_{x : A}B(x)}}}
    };
  {\EqTerm \Gamma f g {\Pi_{x : A}B(x)}}}
\end{gather*}
