\newcommand\Jeq{\ensuremath{\dot{=}}}
\newcommand\Eq[2]{\ensuremath{#1\, \Jeq\, #2}}
\newcommand\EqJ[4]{\ensuremath{#1 \vdash \Eq {#2} {#3} : #4}}
\newcommand\Type[2]{\ensuremath{#1 \vdash #2 \; \text{type}}}
\newcommand\TypeJ[3]{\ensuremath{#1 \vdash #2 : #3}}

\begin{questions}
  \question Give a derivation for the following \textbf{element conversion rule}:
  \[ 
    \ib{\irule
      {\Type{\Gamma}{\Eq {A} {A'}}}
      {\TypeJ \Gamma a A};
      {\TypeJ {\Gamma} {a} {A'}}}
  \]
  \question Give a derivation for the following \textbf{congruence rule} for element conversion:
  \[ 
    \ib{\irule
      {\Type \Gamma {\Eq {A} {A'}}}
      {\EqJ \Gamma {a} {b} {A}}; 
      {\EqJ \Gamma {a} {b} {A'}}
    }
  \]
\end{questions}

\subsubsection*{Solution \#1}

\begin{gather*}
{\small
    \ib{\irule[subst]
      {\Gamma \vdash a : A}
      {\irule[var-conv]
        {\irule[sym]{\Type \Gamma {\Eq {A} {A'}}}; {\Type \Gamma {\Eq {A'} {A}}}}
        {\irule[var]{};{\Gamma, x : A' \vdash x : A'}}; 
        {\Gamma , x : A \vdash x : A'}};
      {\TypeJ {\Gamma} {x[a/x]} {A'[a/x]}}}} 
\end{gather*}
